MedVault Frontend - API Contract & Requirements

PROJECT: React/TypeScript healthcare encryption platform. Backend: http://localhost:8001/api/v1, Frontend: port 3001
STACK: React 18, TypeScript, Vite, TailwindCSS, Zustand

CRITICAL: ALL encryption/decryption is CLIENT-SIDE. Backend never sees plaintext. Use Web Crypto API (AES-256-GCM).

=== KEY MANAGEMENT ===
- Patient: 32-byte master key derived from password, wrap with PBKDF2
- Login: Unwrap master key, store in memory only (NEVER localStorage)
- Encryption: AES-256-GCM with 12-byte nonce
- Doctor Access: Patient encrypts master key with doctor's X25519 public key

=== CRYPTO (TypeScript) ===

export class CryptoService {
  static generateRandomBytes(length: number): Uint8Array {
    return crypto.getRandomValues(new Uint8Array(length));
  }

  static async deriveKeyFromPassword(password: string, salt: Uint8Array, iterations = 100000): Promise<CryptoKey> {
    const keyMaterial = await crypto.subtle.importKey('raw', new TextEncoder().encode(password), 'PBKDF2', false, ['deriveKey']);
    return crypto.subtle.deriveKey({name:'PBKDF2',salt,iterations,hash:'SHA-256'}, keyMaterial, {name:'AES-GCM',length:256}, true, ['encrypt','decrypt']);
  }

  static async encryptAESGCM(data: Uint8Array, key: CryptoKey): Promise<{nonce: Uint8Array, ciphertext: Uint8Array}> {
    const nonce = this.generateRandomBytes(12);
    const ciphertext = await crypto.subtle.encrypt({name:'AES-GCM',iv:nonce}, key, data);
    return {nonce, ciphertext: new Uint8Array(ciphertext)};
  }

  static async decryptAESGCM(ciphertext: Uint8Array, nonce: Uint8Array, key: CryptoKey): Promise<Uint8Array> {
    const plaintext = await crypto.subtle.decrypt({name:'AES-GCM',iv:nonce}, key, ciphertext);
    return new Uint8Array(plaintext);
  }
}

// Encrypt image
const encryptImage = async (file: File, masterKey: CryptoKey) => {
  const data = new Uint8Array(await file.arrayBuffer());
  const {nonce, ciphertext} = await CryptoService.encryptAESGCM(data, masterKey);
  const combined = new Uint8Array(nonce.length + ciphertext.length);
  combined.set(nonce, 0);
  combined.set(ciphertext, nonce.length);
  return new Blob([combined], {type:'application/octet-stream'});
};

// Decrypt image
const decryptImage = async (encryptedBlob: Blob, masterKey: CryptoKey) => {
  const data = new Uint8Array(await encryptedBlob.arrayBuffer());
  const nonce = data.slice(0, 12);
  const ciphertext = data.slice(12);
  const plaintext = await CryptoService.decryptAESGCM(ciphertext, nonce, masterKey);
  return new Blob([plaintext], {type:'image/jpeg'});
};

=== NOTES ===
- CORS: Backend allows localhost:3001, 3000, 5173
- Errors: {"detail":"Error message"}
- Master Key: Memory only, never persist
- Token Refresh: Auto-refresh before 15min expiry
- Image Display: URL.createObjectURL(), remember revoke
- Description Format: base64(12-byte-nonce + encrypted-text), min 13 bytes before encoding
- File Uploads: Use FormData with multipart/form-data
